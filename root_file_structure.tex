\section*{Implementation outline}
\begin{itemize}[leftmargin=*]
\item \textbf{Data ingestion}
    \begin{itemize}
    \item Source files: output-enriched-llama3.3 - Sheet1.csv, \texttt{lightning\_ai\_metadata/input.csv}, \texttt{results/compare/*.csv}
    \item Orchestration script: \texttt{lightning\_ai\_metadata/main.py}
    \end{itemize}
\item \textbf{Preprocessing}
    \begin{itemize}
    \item Normalize CSV formats, clean text fields, deduplicate rows.
    \item DOI/title reconciliation using \texttt{results/compare/title\_match\_fallback.py}.
    \end{itemize}
\item \textbf{Analysis}
    \begin{itemize}
    \item Alignment summaries and DOI matching: \texttt{results/compare/build\_alignment\_summary.py}.
    \item Topic modeling pipeline in \texttt{results/topic-modeling/bertopic/} (notebooks and model-specific outputs).
    \end{itemize}
\item \textbf{Outputs}
    \begin{itemize}
    \item CSV summaries in \texttt{results/compare/} and topic summaries in each model folder.
    \item Figures and UMAP embeddings in \texttt{results/topic-modeling/bertopic/.../figures} and corresponding numpy files.
    \end{itemize}
\item \textbf{Suggested next steps}
    \begin{enumerate}
    \item Add a README explaining how to run \texttt{lightning\_ai\_metadata/main.py} and the BERTopic notebooks.
    \item Add a \texttt{run.sh} or \texttt{Makefile} to standardize pipeline execution.
    \item Add unit tests for \texttt{title\_match\_fallback.py} and \texttt{build\_alignment\_summary.py}.
    \end{enumerate}
\end{itemize}
